CREATE NONCLUSTERED INDEX IX_TXINV_YTN 
ON dbo.TXINV ([YEAR], [TYPE], [NAME])
INCLUDE (
  [ICODE], [LIST#], [DIST], [PHASE], [TAXT],
  [PDAT], [CCNO], [CDATE], [CCETAX], [SUSCD], [SUSDT])
  go

  CREATE NONCLUSTERED INDEX IX_TXHST_LYT 
ON dbo.TXHST ([list#], [year], [type], [pdate])
INCLUDE ([PAMT], [LAMT], [IAMT], [PCAMT], [RCODE], [ADJCD], [BATCHA], [SUSCD]);
go

CREATE NONCLUSTERED INDEX IX_TXCOEA_LTYC_Descending
ON dbo.TXCOEA ([LIST#], [TYPE], [YEAR], [CHDATE] DESC, [CHTIME] DESC)
INCLUDE ([CDATE], [CETAX]);
go

CREATE NONCLUSTERED INDEX IX_UTCOEA_LTYC_Descending
ON dbo.UTCOEA (
  [LIST#], 
  [TYPE], 
  [YEAR], 
  [CHDATE] DESC, 
  [CHTIME] DESC
)
INCLUDE (
  [CDATE], 
  [CETAX]
);
go
