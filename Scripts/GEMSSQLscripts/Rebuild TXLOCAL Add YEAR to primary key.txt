CREATE TABLE [dbo].[TXLOCAL_NEW](
    [LIST#] [numeric](6, 0) NOT NULL,
    [YEAR] [numeric](4, 0) NOT NULL,
    [TYPE] [char](1) NOT NULL,
    [BENCDE] [char](2) NOT NULL,
    [BENAMT] [numeric](7, 2) NOT NULL,

PRIMARY KEY CLUSTERED
(
    [LIST#] ASC,
    [YEAR] ASC,
    [TYPE] ASC,
    [BENCDE] ASC
)
WITH (
    PAD_INDEX = OFF,
    STATISTICS_NORECOMPUTE = OFF,
    IGNORE_DUP_KEY = OFF,
    ALLOW_ROW_LOCKS = ON,
    ALLOW_PAGE_LOCKS = ON
) ON [PRIMARY]
) ON [PRIMARY]
GO


SET ANSI_PADDING ON
GO


/****** Copy existing data into new table ******/
INSERT INTO [dbo].[TXLOCAL_NEW]
(
    [LIST#],
    [YEAR],
    [TYPE],
    [BENCDE],
    [BENAMT]
)
SELECT
    [LIST#],
    0 AS [YEAR],
    [TYPE],
    [BENCDE],
    [BENAMT]
FROM [dbo].[TXLOCAL]
GO

    ALLOW_ROW_LOCKS = ON,
