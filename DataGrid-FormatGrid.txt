    Public Sub FormatGrid()

    Dim myGridTableStyle As DataGridTableStyle = New DataGridTableStyle

    Call ShowGrid()
    myGridTableStyle.MappingName = ds.Tables(0).TableName

    Try
      DataGrdList.TableStyles.RemoveAt(0)
    Catch
    End Try

    Try
      DataGrdList.TableStyles.Add(myGridTableStyle)
    Catch
      Exit Sub
    End Try

      If RbName.Checked Or RbBalances.Checked Then
        GridName(myGridTableStyle)
      End If
      If RbLoc.Checked Then
        GridLocName(myGridTableStyle)
      End If
      If RbReg.Checked Then
        GridRegno(myGridTableStyle)
      End If

    End Sub
    Public Sub ShowGrid()
       Cursor.Current = Cursors.WaitCursor
       If RbName.Checked Then
        ds = mytxinvl8.GetViewbyName(TxtPos.Text, 50)
       End If
       If RbLoc.Checked Then
        ds = mytxinvla.GetViewbyLoc(TxtPos.Text, TxtPosNo.Text, 50)
       End If
       If RbReg.Checked Then
        ds = mytxinvlc.GetViewbyRegNo(TxtPos.Text, 50)
       End If
       If RbBalances.Checked Then
        ds = mytxinvlh.GetViewbyName(TxtPos.Text, 50)
       End If

      DataGrdList.DataSource = ds.Tables(0)
      DataGrdList.Refresh()
      Cursor.Current = Cursors.Default

    End Sub
