DECLARE @ListNum INT = 110110;
UPDATE a SET 
    a.tax1  = b.tax1,
    a.tax2  = b.tax2,
    a.tx3rd  = b.tx3rd,
    a.taxt   = b.tax1 + b.tax2 + b.tx3rd + a.tx4th, 
    a.payrec = b.payrec,
    a.bald   = (b.tax1 + b.tax2 + b.tx3rd + a.tx4th) - b.payrec
FROM txinv AS a 
LEFT JOIN southub..muninv AS b 
    ON a.list# = b.list# 
   AND a.type = b.type 
   AND a.year = b.year 
WHERE a.list# = @ListNum 
  AND a.type = 'D' 
  AND a.year = 2026;


DECLARE @MinRecid INT = 32140;
DECLARE @MaxRecid INT = 32142;
DECLARE @ListNum  INT = 200204;
DECLARE @StartRecid INT = 3372708;


INSERT INTO txhst 
SELECT * FROM southub..munhst 
WHERE recid >= @MinRecid 
  AND recid <= @MaxRecid 
  AND [list#] = @ListNum;

WITH SequentialRows AS (
    SELECT 
        recid,
        ROW_NUMBER() OVER (ORDER BY recid ASC) AS RowNum
    FROM txhst
    WHERE recid >= @MinRecid 
      AND recid <= @MaxRecid 
      AND [list#] = @ListNum
)
UPDATE SequentialRows
SET recid = @StartRecid + RowNum;
